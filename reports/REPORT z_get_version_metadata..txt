REPORT z_get_version_metadata.

*******************************************************
* Seleção
*******************************************************
DATA: gv_dummy_file TYPE sy-repid,
      gv_dummy_cat  TYPE char20.

CONSTANTS: gc_path TYPE string VALUE 'C:\temp\versions_metadata.txt'.

SELECT-OPTIONS:
  p_file FOR gv_dummy_file OBLIGATORY,
  p_cat  FOR gv_dummy_cat  OBLIGATORY.

*******************************************************
* Tipos
*******************************************************
TYPES: BEGIN OF ty_response,
         id        TYPE string,
         title     TYPE string,
         updated   TYPE string,
         request   TYPE string,
         file_name TYPE string,
         category  TYPE string,
       END OF ty_response.

TYPES: tt_response   TYPE STANDARD TABLE OF ty_response WITH DEFAULT KEY.

TYPES: BEGIN OF ty_json_line,
         line TYPE string,
       END OF ty_json_line.

TYPES: tt_json_lines TYPE STANDARD TABLE OF ty_json_line WITH DEFAULT KEY.

*******************************************************
* Serviço de versões
*******************************************************
CLASS lcl_version_service DEFINITION FINAL.
  PUBLIC SECTION.
    CLASS-METHODS get_versions_multi
      RETURNING
        VALUE(rt_result) TYPE tt_response.

    CLASS-METHODS get_versions_single
      IMPORTING
        iv_file_name     TYPE sy-repid
        iv_category      TYPE char20
      RETURNING
        VALUE(rt_result) TYPE tt_response.

  PRIVATE SECTION.
    CLASS-METHODS normalize_category
      IMPORTING
        iv_category        TYPE char20
      RETURNING
        VALUE(rv_category) TYPE char4.

    CLASS-METHODS format_iso8601
      IMPORTING
        iv_date          TYPE sy-datum
        iv_time          TYPE sy-uzeit
      RETURNING
        VALUE(rv_result) TYPE string.

    CLASS-METHODS category_to_label
      IMPORTING
        iv_category       TYPE char4
      RETURNING
        VALUE(rv_label)   TYPE string.

    CLASS-METHODS read_versions_program
      IMPORTING
        iv_prog_name     TYPE sy-repid
        iv_category      TYPE char4
      RETURNING
        VALUE(rt_result) TYPE tt_response.

ENDCLASS.

CLASS lcl_version_service IMPLEMENTATION.

  METHOD get_versions_multi.
    DATA: lt_result_single TYPE tt_response,
          ls_file          LIKE LINE OF p_file,
          ls_cat           LIKE LINE OF p_cat.

    CLEAR rt_result.

    LOOP AT p_file INTO ls_file.
      LOOP AT p_cat INTO ls_cat.
        lt_result_single = lcl_version_service=>get_versions_single(
          iv_file_name = ls_file-low
          iv_category  = ls_cat-low ).

        IF lt_result_single IS NOT INITIAL.
          APPEND LINES OF lt_result_single TO rt_result.
        ENDIF.
      ENDLOOP.
    ENDLOOP.

    SORT rt_result BY file_name
                      category
                      updated DESCENDING
                      request
                      id.

  ENDMETHOD.

  METHOD normalize_category.
    DATA: lv_category TYPE char20.

    CLEAR rv_category.

    lv_category = iv_category.
    TRANSLATE lv_category TO UPPER CASE.

    CASE lv_category.
      WHEN 'PROGRAM'         OR 'PROG' OR 'REPORT'. rv_category = 'PROG'.
      WHEN 'CLASS'           OR 'CLAS'.             rv_category = 'CLAS'.
      WHEN 'ENHANCEMENT'     OR 'ENHO'.             rv_category = 'ENHO'.
      WHEN 'FUNCTION_MODULE' OR 'FUNC'.             rv_category = 'FUNC'.
      WHEN 'FUNCTION_GROUP'  OR 'FUGR'.             rv_category = 'FUGR'.
      WHEN 'INDEX'           OR 'INDX'.             rv_category = 'INDX'.
      WHEN 'INTERFACE'       OR 'INTF'.             rv_category = 'INTF'.
      WHEN 'TABLE'           OR 'TABL'.             rv_category = 'TABL'.
      WHEN 'CDS_VIEW'        OR 'DDLS'.             rv_category = 'DDLS'.
      WHEN 'INCLUDE'         OR 'INCL'.             rv_category = 'INCL'.
      WHEN 'DOMAIN'          OR 'DOMA'.             rv_category = 'DOMA'.
      WHEN 'DATA_ELEMENT'    OR 'DTEL'.             rv_category = 'DTEL'.
      WHEN OTHERS.
        CLEAR rv_category.
    ENDCASE.
  ENDMETHOD.

  METHOD get_versions_single.
    DATA: lv_category TYPE char4.

    CLEAR rt_result.

    lv_category = normalize_category( iv_category ).

    IF lv_category IS INITIAL.
      RETURN.
    ENDIF.

    CASE lv_category.
      WHEN 'PROG'.
        rt_result = lcl_version_service=>read_versions_program(
          iv_prog_name = iv_file_name
          iv_category  = lv_category ).
      WHEN OTHERS.
        " TODO: implementar leitura para CLAS, ENHO, FUNC, FUGR,
        " INDX, INTF, TABL, DDLS, INCL, DOMA, DTEL
        CLEAR rt_result.
    ENDCASE.
  ENDMETHOD.

  METHOD category_to_label.
    CASE iv_category.
      WHEN 'PROG'. rv_label = 'Report'.
      WHEN 'CLAS'. rv_label = 'Classe'.
      WHEN 'ENHO'. rv_label = 'Enhancement'.
      WHEN 'FUNC'. rv_label = 'Módulo de função'.
      WHEN 'FUGR'. rv_label = 'Grupo de funções'.
      WHEN 'INDX'. rv_label = 'Índice'.
      WHEN 'INTF'. rv_label = 'Interface'.
      WHEN 'TABL'. rv_label = 'Tabela'.
      WHEN 'DDLS'. rv_label = 'CDS View'.
      WHEN 'INCL'. rv_label = 'Include'.
      WHEN 'DOMA'. rv_label = 'Domínio'.
      WHEN 'DTEL'. rv_label = 'Elemento de dados'.
      WHEN OTHERS. rv_label = iv_category.
    ENDCASE.
  ENDMETHOD.

  METHOD format_iso8601.
    rv_result = iv_date+0(4) && '-' && iv_date+4(2) && '-' && iv_date+6(2) &&
                'T' && iv_time+0(2) && ':' && iv_time+2(2) && ':' && iv_time+4(2).
  ENDMETHOD.

  METHOD read_versions_program.
    DATA: ls_response  TYPE ty_response,
          lv_prog_name TYPE sy-repid,
          lv_count     TYPE i,
          lv_udat      TYPE reposrc-udat,
          lv_utime     TYPE reposrc-utime,
          lv_versno    TYPE i,
          lv_visual    TYPE n LENGTH 5.

    TYPES: BEGIN OF ty_join,
             versno  TYPE vrsd-versno,
             korrnum TYPE vrsd-korrnum,
             datum   TYPE vrsd-datum,
             zeit    TYPE vrsd-zeit,
             as4text TYPE e07t-as4text,
           END OF ty_join.

    DATA: lt_dados TYPE STANDARD TABLE OF ty_join,
          ls_dados TYPE ty_join.

    CLEAR rt_result.

    lv_prog_name = iv_prog_name.
    TRANSLATE lv_prog_name TO UPPER CASE.

    " Versões históricas ordenadas da mais ANTIGA para a mais NOVA
    " (o número visual será atribuído nessa ordem: 1, 2, 3...)
    SELECT vrsd~versno vrsd~korrnum vrsd~datum vrsd~zeit e07t~as4text
      INTO CORRESPONDING FIELDS OF TABLE lt_dados
      FROM vrsd
      LEFT OUTER JOIN e07t ON e07t~trkorr = vrsd~korrnum
                          AND e07t~langu  = sy-langu
      WHERE vrsd~objtype = 'REPS'
        AND vrsd~objname = lv_prog_name
      ORDER BY vrsd~datum ASCENDING
               vrsd~zeit  ASCENDING.

    " Atribui número visual (1, 2, 3...) e depois inverte para DESC
    lv_versno = 0.
    LOOP AT lt_dados INTO ls_dados.
      lv_versno = lv_versno + 1.
      lv_visual = lv_versno.

      CLEAR ls_response.
      ls_response-id        = lv_visual.
      ls_response-title     = ls_dados-as4text.
      ls_response-updated   = format_iso8601( iv_date = ls_dados-datum iv_time = ls_dados-zeit ).
      ls_response-request   = ls_dados-korrnum.
      ls_response-file_name = lv_prog_name.
      ls_response-category  = category_to_label( iv_category ).
      APPEND ls_response TO rt_result.
    ENDLOOP.

    " Ativo fica por cima, como 00000
    SELECT SINGLE udat utime
      FROM reposrc
      INTO (lv_udat, lv_utime)
      WHERE progname = lv_prog_name
        AND r3state  = 'A'.

    IF sy-subrc = 0.
      CLEAR ls_response.
      ls_response-id        = '00000'.
      ls_response-title     = 'Versão ativa'.
      ls_response-file_name = lv_prog_name.
      ls_response-category  = category_to_label( iv_category ).
      ls_response-request   = ''.
      ls_response-updated   = format_iso8601( iv_date = lv_udat iv_time = lv_utime ).
      INSERT ls_response INTO rt_result INDEX 1.
    ENDIF.

    " Reordena: ativo no topo, depois histórico do mais novo ao mais antigo
    SORT rt_result BY id DESCENDING.

    " Garante que 00000 (ativo) fique no topo apesar do sort
    " (como '00000' é numericamente o menor, precisamos trazê-lo pra frente)
    LOOP AT rt_result INTO ls_response WHERE id = '00000'.
      DELETE rt_result INDEX sy-tabix.
      INSERT ls_response INTO rt_result INDEX 1.
      EXIT.
    ENDLOOP.

  ENDMETHOD.

ENDCLASS.

*******************************************************
* Escrita do arquivo JSON no Frontend
*******************************************************
CLASS lcl_file_writer DEFINITION FINAL.
  PUBLIC SECTION.
    CLASS-METHODS write_json_file
      IMPORTING
        it_result TYPE tt_response.
ENDCLASS.

CLASS lcl_file_writer IMPLEMENTATION.
  METHOD write_json_file.
    DATA: lv_json  TYPE string,
          lt_lines TYPE tt_json_lines,
          ls_line  TYPE ty_json_line.

    CLEAR lt_lines.

    lv_json = /ui2/cl_json=>serialize(
      data        = it_result
      compress    = abap_false
      pretty_name = /ui2/cl_json=>pretty_mode-camel_case ).

    ls_line-line = lv_json.
    APPEND ls_line TO lt_lines.

    CALL METHOD cl_gui_frontend_services=>gui_download
      EXPORTING
        filename = gc_path
        filetype = 'ASC'
      CHANGING
        data_tab = lt_lines
      EXCEPTIONS
        OTHERS   = 1.

  ENDMETHOD.
ENDCLASS.

*******************************************************
* Execução
*******************************************************
START-OF-SELECTION.
  DATA: lt_result TYPE tt_response.

  lt_result = lcl_version_service=>get_versions_multi( ).

  lcl_file_writer=>write_json_file(
    it_result = lt_result ).

  MESSAGE 'Arquivo gerado em C:\temp\versions_metadata.txt' TYPE 'S'.

END-OF-SELECTION.
